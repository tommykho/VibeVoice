# Runtime dependencies for the VibeVoice CLI and the streaming 0.5B model.
#
#   pip install -r requirements.txt
#
# This is the minimum vibevoice.py, batch_samples.py and
# demo/realtime_model_inference_from_file.py actually import. The gradio /
# websocket / ASR demos need more: pip install -r requirements-demos.txt
#
# `pip install -e .[streamingtts]` installs the same set and additionally puts
# the vibevoice package on the path, which every entry point needs -- run that
# too, or set PYTHONPATH to the repository root.
#
# What a bare `torch` resolves to depends on the platform: on Linux it is a
# CUDA build that bundles several GB of NVIDIA libraries, on Windows a CPU-only
# build. Install torch explicitly FIRST if you want the other one -- the rest of
# this file then finds the dependency satisfied and leaves it alone.
#
#   CPU only:  pip install "torch<2.10" --index-url https://download.pytorch.org/whl/cpu
#   NVIDIA:    pip install "torch<2.10" --index-url https://download.pytorch.org/whl/cu128
#
# Carry the <2.10 pin on that command too. Without it pip takes the newest
# torch, and the downgrade this file then applies strands any companion package
# already installed against the newer one (torchvision pins torch exactly).
# Neither torchvision nor torchaudio is used here -- do not install them.
#
# ffmpeg must be on PATH for audio/video handling; it is not a pip package.

# torch 2.10 tightened its weights-only unpickler to an exact type check, which
# rejects the BaseModelOutputWithPast inside every demo/voices/**/*.pt preset:
# "Can only SETITEMS for dict, collections.OrderedDict, collections.Counter".
torch<2.10

# 4.51.3 exactly: the streaming 0.5B model does not load on later releases.
transformers==4.51.3

accelerate      # transformers needs it for device_map= loading
diffusers       # vibevoice/schedule/dpm_solver.py
librosa         # audio processors; pulls soundfile, scipy, numba, llvmlite
numpy
tqdm
